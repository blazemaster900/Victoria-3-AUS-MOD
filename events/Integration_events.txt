namespace = Integration_events


#natives
Integration_events.1 = {
	type = country_event
	hidden = yes

	trigger = { is_player = yes }

	immediate = {
		#country primal culture integration
		every_country = {
			integrate_native_culture_custom_1 = yes
		}
		#country pop& home integration
		every_state = {
			integrate_native_homeland_custom_1 = yes
			integrate_native_pop_custom_1 = yes
		}
	}
}
